#!/bin/bash -v

# Cloud-init for mini PC / physical server
# This script bootstraps a physical Ubuntu server with:
# - SSH hardening and user setup (franky user)
# - Firewall (nftables)
# - WireGuard VPN client (connects to remote WireGuard server)
# - WireGuard TCP tuning (BBR congestion control, MSS clamping, MTU probing)
# - Franky AI coding agent
# - Development tools (Go, Zig)
# - fail2ban intrusion prevention
# - Docker (optional)
# - Automatic security updates

export DEBIAN_FRONTEND=noninteractive

# ============================================
# Helper: systemctl fallback for containers without systemd PID 1
# ============================================
sctl() {
    if [ -d /run/systemd/system ]; then
        systemctl "$@"
    else
        # Fallback: map common systemctl commands to direct service actions
        action="$1"; shift
        case "$action" in
            enable)
                # no-op without systemd
                ;;
            start|restart|reload)
                service="$1"
                if [ "$service" = "ssh" ]; then service="ssh"; fi
                service "$service" "$action" 2>/dev/null || true
                ;;
            is-active)
                service="$1"
                service "$service" status 2>/dev/null && echo active || echo inactive
                ;;
            *)
                systemctl "$action" "$@" 2>/dev/null || true
                ;;
        esac
    fi
}

# ============================================
# Configuration (override via template variables)
# ============================================
MINIPC_USER="{{ MINIPC_USER }}"
MINIPC_SSH_PORT="{{ MINIPC_SSH_PORT }}"
MINIPC_DOCKER="{{ MINIPC_DOCKER }}"
MINIPC_NIC="{{ MINIPC_NIC }}"
FRANKY_VERSION="{{ FRANKY_VERSION }}"
FRANKY_BOX_VERSION="{{ FRANKY_BOX_VERSION }}"

# Set defaults for empty template variables
[ -z "$MINIPC_USER" ] && MINIPC_USER="frank.ittermann"
[ -z "$MINIPC_SSH_PORT" ] && MINIPC_SSH_PORT="22"
[ -z "$MINIPC_DOCKER" ] && MINIPC_DOCKER="true"
[ -z "$FRANKY_VERSION" ] && FRANKY_VERSION="0.47.0"
[ -z "$FRANKY_BOX_VERSION" ] && FRANKY_BOX_VERSION="0.5.1"
if [ -z "$MINIPC_NIC" ]; then
    # Try iproute2 first, fall back to /sys or 'eth0' for minimal containers
    if command -v ip >/dev/null 2>&1; then
        MINIPC_NIC="$(ip route | grep default | awk '{print $5}' | head -1)"
    elif [ -d /sys/class/net ]; then
        # Pick the first non-loopback interface
        MINIPC_NIC="$(ls /sys/class/net | grep -v lo | head -1)"
    fi
    [ -z "$MINIPC_NIC" ] && MINIPC_NIC="eth0"
fi

# ============================================
# System Update
# ============================================
echo "=== Updating system ==="
apt-get update -y
apt-get upgrade -y
apt-get dist-upgrade -y

# ============================================
# Base Packages
# ============================================
echo "=== Installing base packages ==="
apt-get install -y \
    apt-transport-https \
    ca-certificates \
    curl \
    fail2ban \
    gnupg2 \
    jq \
    lsb-release \
    pwgen \
    software-properties-common \
    ubuntu-keyring \
    wireguard-tools

# ============================================
# SSH User Setup
# ============================================
echo "=== Setting up SSH user ==="
useradd -m -s /bin/bash -G sudo "$MINIPC_USER" 2>/dev/null || true
mkdir -p "/home/$MINIPC_USER/.ssh/"
mkdir -p "/home/$MINIPC_USER/github/"
echo "$MINIPC_USER ALL=(ALL) NOPASSWD:ALL" >"/etc/sudoers.d/$MINIPC_USER"
chmod 0440 "/etc/sudoers.d/$MINIPC_USER"

# Copy root's authorized keys if they exist (for Hetzner/cloud-init provisioned machines)
if [ ! -s "/home/$MINIPC_USER/.ssh/authorized_keys" ] && [ -f /root/.ssh/authorized_keys ]; then
    cp /root/.ssh/authorized_keys "/home/$MINIPC_USER/.ssh/authorized_keys"
fi
[ -f "/home/$MINIPC_USER/.ssh/authorized_keys" ] && chmod 600 "/home/$MINIPC_USER/.ssh/authorized_keys"
chown -R "$MINIPC_USER:$MINIPC_USER" "/home/$MINIPC_USER"

# ============================================
# SSH Hardening
# ============================================
echo "=== Hardening SSH ==="
sed -i -e '/^PermitRootLogin/s/^.*$/PermitRootLogin no/' /etc/ssh/sshd_config
sed -i 's|[#]*PasswordAuthentication yes|PasswordAuthentication no|g' /etc/ssh/sshd_config
if ! grep -q "PubkeyAcceptedAlgorithms" /etc/ssh/sshd_config; then
    echo "PubkeyAcceptedAlgorithms +ssh-rsa" >> /etc/ssh/sshd_config
fi
if ! grep -q "AllowUsers" /etc/ssh/sshd_config; then
    echo "AllowUsers $MINIPC_USER" >> /etc/ssh/sshd_config
fi

# Ensure a valid Port directive (also fixes corruption from previous runs
# that may have written a literal placeholder like {{ MINIPC_SSH_PORT }}).
if grep -q '^Port ' /etc/ssh/sshd_config; then
    sed -i "s/^Port .*/Port $MINIPC_SSH_PORT/" /etc/ssh/sshd_config
elif grep -q '^#*Port ' /etc/ssh/sshd_config; then
    sed -i "s/^#*Port .*/Port $MINIPC_SSH_PORT/" /etc/ssh/sshd_config
elif [ "$MINIPC_SSH_PORT" != "22" ]; then
    echo "Port $MINIPC_SSH_PORT" >> /etc/ssh/sshd_config
fi

# Validate sshd config before restarting; fall back to restart (more reliable than reload in containers)
if command -v sshd >/dev/null 2>&1 && sshd -t 2>/dev/null; then
    sctl restart ssh.service
else
    echo "WARNING: sshd config invalid or sshd not found, skipping restart"
fi

# ============================================
# Firewall Setup (nftables)
# ============================================
echo "=== Setting up nftables firewall ==="
apt-get install -y nftables

cat > /etc/nftables.conf <<- 'NFTABLES'
#!/usr/sbin/nft -f
# Mini PC firewall rules

flush ruleset

table inet filter {
    chain input {
        type filter hook input priority 0; policy drop;

        # Allow loopback
        iif "lo" accept

        # Allow established/related connections
        ct state established,related accept
        ct state invalid drop

        # Allow ICMP (ping)
        ip protocol icmp accept
        ip6 nexthdr icmpv6 accept

        # Allow SSH
        tcp dport { 22 } accept

        # Allow WireGuard client port (for outgoing connections)
        udp sport { 51820 } accept

        # Allow HTTP/HTTPS
        tcp dport { 80, 443 } accept

        # Allow DNS (responses from DNS servers)
        udp sport { 53 } accept
        tcp sport { 53 } accept

        # Allow mDNS for local network discovery
        udp dport { 5353 } accept

        # Allow franky web UI
        tcp dport { 8787 } accept

        # Allow franky-box inbox/outbox API
        tcp dport { 8080 } accept

        # Log dropped packets
        log prefix "nftables-drop: " drop
    }

    chain forward {
        type filter hook forward priority 0; policy drop;

        # Allow established/related
        ct state established,related accept

        # Log dropped packets
        log prefix "nftables-fwd-drop: " drop
    }

    chain output {
        type filter hook output priority 0; policy accept;
    }
}

# MSS clamping for WireGuard tunnel (mtu 1420 < ethernet 1500).
# Prevents PMTU blackholes that stall HTTP/SSE/bulk transfers over the VPN.
table ip mangle {
    chain postrouting {
        type filter hook postrouting priority mangle; policy accept;
        oifname "wg0" tcp flags syn / syn,rst counter tcp option maxseg size set rt mtu
    }
}
NFTABLES

# Update SSH port in nftables if non-default
if [ "$MINIPC_SSH_PORT" != "22" ]; then
    sed -i "s/tcp dport { 22 }/tcp dport { $MINIPC_SSH_PORT }/" /etc/nftables.conf
fi

sctl enable nftables
# Apply firewall rules only on real servers, not in Docker/Vagrant containers.
# In containers, conntrack is often unavailable so 'ct state established,related'
# fails to load, but 'policy drop' still takes effect — blocking DNS responses
# and all subsequent network access. The config file is still written for real
# server use, but not applied in containers.
if [ -f /.dockerenv ] || grep -qE 'docker|containerd|kubepods' /proc/1/cgroup 2>/dev/null || [ -n "${container+x}" ]; then
    echo "WARNING: running in a container, skipping nftables rule application"
else
    sctl restart nftables || echo "WARNING: nftables restart failed"
fi

# ============================================
# WireGuard TCP Tuning (BBR + MTU probing)
# ============================================
echo "=== Tuning TCP for WireGuard (BBR + MTU probing) ==="

# Load tcp_bbr now and at boot; fall back silently if unavailable.
modprobe tcp_bbr 2>/dev/null || true
echo "tcp_bbr" > /etc/modules-load.d/tcp_bbr.conf

# BBR beats CUBIC on high-RTT, high-jitter tunnels: it models bandwidth+RTT
# instead of reacting to loss, so bursts drain in ~1 RTT instead of stuttering.
# MTU probing self-heals PMTU blackholes (DF packets that vanish without ICMP).
cat > /etc/sysctl.d/99-wg-bbr-mtu.conf <<'EOF'
net.ipv4.tcp_congestion_control = bbr
net.ipv4.tcp_mtu_probing = 1
EOF
sysctl --system >/dev/null 2>&1

# ============================================
# fail2ban
# ============================================
echo "=== Setting up fail2ban ==="

sctl enable fail2ban
sctl restart fail2ban

# ============================================
# Install Franky
# ============================================
echo "=== Installing franky v${FRANKY_VERSION} ==="
cd /tmp
ARCH=$(uname -m)
case "$ARCH" in
    x86_64)  FRANKY_ARCH="amd64" ;;
    aarch64) FRANKY_ARCH="arm64" ;;
    armv7l)   FRANKY_ARCH="arm" ;;
    *)        FRANKY_ARCH="amd64" ;;
esac
curl -OL "https://github.com/fr12k/franky/releases/download/v${FRANKY_VERSION}/franky_linux_${FRANKY_ARCH}"
mv "franky_linux_${FRANKY_ARCH}" /usr/bin/franky
chmod +x /usr/bin/franky

# Verify it's a valid ELF binary
head -c 4 /usr/bin/franky | od -An -tx1 | grep -q '7f 45 4c 46' && echo "valid ELF binary" || echo "WARNING: not an ELF binary"

echo "=== franky installation complete ==="

# ============================================
# Install franky-box
# ============================================
echo "=== Installing franky-box v${FRANKY_BOX_VERSION} ==="
cd /tmp
ARCH=$(uname -m)
case "$ARCH" in
    x86_64)  BOX_ARCH="amd64" ;;
    aarch64)  BOX_ARCH="arm64" ;;
    armv7l)   BOX_ARCH="arm" ;;
    *)        BOX_ARCH="amd64" ;;
esac
curl -OL "https://github.com/fr12k/franky-box/releases/download/v${FRANKY_BOX_VERSION}/franky-box_linux_${BOX_ARCH}"
mv "franky-box_linux_${BOX_ARCH}" /usr/bin/franky-box
chmod +x /usr/bin/franky-box

# Verify it's a valid ELF binary
head -c 4 /usr/bin/franky-box | od -An -tx1 | grep -q '7f 45 4c 46' && echo "valid ELF binary" || echo "WARNING: not an ELF binary"

echo "=== franky-box installation complete ==="

# ============================================
# franky-box Systemd Service
# ============================================
echo "=== Creating franky-box systemd service ==="

cat > /etc/systemd/system/franky-box.service <<EOF
[Unit]
Description=franky-box — embedded sqlite task inbox/outbox queue
After=network.target
StartLimitIntervalSec=120s
StartLimitBurst=10

[Service]
Environment="FRANKY_BOX_ADMIN_TOKEN=${FRANKY_BOX_ADMIN_TOKEN:-admin-token-change-me}"
WorkingDirectory=/home/$MINIPC_USER/.franky-box
ExecStart=/usr/bin/franky-box
Restart=on-failure
RestartSec=10s

[Install]
WantedBy=multi-user.target
EOF

# Create franky-box data directory
mkdir -p /home/$MINIPC_USER/.franky-box

# ============================================
# Install Development Tools (Go, Zig)
# ============================================
echo "=== Installing tools (zig, golang) ==="

# Detect architecture for binary downloads
ARCH=$(uname -m)
case "$ARCH" in
    x86_64)  ZIG_ARCH="x86_64"; GO_ARCH="amd64" ;;
    aarch64)  ZIG_ARCH="aarch64"; GO_ARCH="arm64" ;;
    armv7l)   ZIG_ARCH="armv7a"; GO_ARCH="arm" ;;
    *)        ZIG_ARCH="x86_64"; GO_ARCH="amd64" ;;
esac

cd /tmp
ZIG_VERSION="0.17.0-dev.1786+75044cb04"
ZIG_URL="https://ziglang.org/builds/zig-${ZIG_ARCH}-linux-${ZIG_VERSION}.tar.xz"
echo "Downloading zig ${ZIG_VERSION}..."
if wget -q "$ZIG_URL"; then
    tar -xf "zig-${ZIG_ARCH}-linux-${ZIG_VERSION}.tar.xz" -C /usr/local/bin --strip-components=1 || echo "WARNING: zig extraction failed"
    rm -f "zig-${ZIG_ARCH}-linux-${ZIG_VERSION}.tar.xz"
else
    echo "WARNING: failed to download zig from $ZIG_URL"
fi

GO_VERSION="1.26.6"
GO_TARBALL="go${GO_VERSION}.linux-${GO_ARCH}.tar.gz"
if wget -q "https://go.dev/dl/${GO_TARBALL}"; then
    rm -rf /usr/local/go
    tar -C /usr/local -xzf "$GO_TARBALL"
    ln -sf /usr/local/go/bin/go /usr/local/bin/go
    ln -sf /usr/local/go/bin/gofmt /usr/local/bin/gofmt
    rm -f "$GO_TARBALL"
else
    echo "WARNING: failed to download go from https://go.dev/dl/${GO_TARBALL}"
fi

echo "=== tools installation complete ==="

# ============================================
# Franky Systemd Service
# ============================================
echo "=== Creating franky systemd service ==="

cat > /etc/systemd/system/franky.service <<EOF
[Unit]
Description=franky
After=network.target
StartLimitIntervalSec=120s
StartLimitBurst=10

[Service]
EnvironmentFile=-/etc/franky.env
WorkingDirectory=/home/$MINIPC_USER/github/
ExecStart=/usr/bin/franky --profile ollama-cloud/glm-5.2:cloud --role full --yes --mode worker --inbox-server \${FRANKY_INBOX_SERVER} --agent-id \${FRANKY_AGENT_ID} --agent-secret \${FRANKY_AGENT_SECRET} --team-id \${FRANKY_TEAM_ID} --web-port 8787
Restart=on-failure
RestartSec=10s

[Install]
WantedBy=multi-user.target
EOF

# ============================================
# Docker (optional)
# ============================================
if [ "$MINIPC_DOCKER" = "true" ]; then
    echo "=== Installing Docker ==="
    curl -fsSL https://get.docker.com | sh -

    # Add user to docker group
    usermod -aG docker "$MINIPC_USER"

    sctl enable docker
    sctl start docker
fi

# ============================================
# Automatic Security Updates
# ============================================
echo "=== Setting up automatic security updates ==="
apt-get install -y unattended-upgrades
dpkg-reconfigure -plow unattended-upgrades

cat > /etc/apt/apt.conf.d/20auto-upgrades <<- EOF
APT::Periodic::Update-Package-Lists "1";
APT::Periodic::Unattended-Upgrade "1";
APT::Periodic::AutocleanInterval "7";
EOF

# ============================================
# Start Services
# ============================================
echo "=== Starting services ==="

# Start fail2ban
sctl start fail2ban

# Start franky-box first (franky registers against it)
sctl start franky-box
sctl enable franky-box

# ============================================
# Register franky agent with franky-box
# ============================================
echo "=== Registering franky agent with franky-box ==="

# Wait for franky-box to be ready (up to 30s)
BOX_READY=false
for i in $(seq 1 30); do
    if curl -s -o /dev/null -w "%{http_code}" http://127.0.0.1:8080/ 2>/dev/null | grep -qE '^2[0-9][0-9]$'; then
        BOX_READY=true
        break
    fi
    sleep 1
done

REG_OK=false
if [ "$BOX_READY" = "true" ]; then
    # Register a new agent via POST /v1/agents
    REG_RESPONSE=$(curl -sf -X POST http://127.0.0.1:8080/v1/agents \
        -H "Authorization: Bearer ${FRANKY_BOX_ADMIN_TOKEN:-admin-token-change-me}" \
        -H "Content-Type: application/json" \
        -d '{}') || REG_RESPONSE=""

    if [ -n "$REG_RESPONSE" ]; then
        # Parse agent_id and agent_secret from the JSON response
        FRANKY_AGENT_ID=$(echo "$REG_RESPONSE" | jq -r '.agent_id // empty')
        FRANKY_AGENT_SECRET=$(echo "$REG_RESPONSE" | jq -r '.agent_secret // empty')
        FRANKY_TEAM_ID=$(echo "$REG_RESPONSE" | jq -r '.team_id // "default"')

        if [ -n "$FRANKY_AGENT_ID" ] && [ -n "$FRANKY_AGENT_SECRET" ]; then
            echo "Registered agent: $FRANKY_AGENT_ID (team: $FRANKY_TEAM_ID)"

            # Ensure env file is secure before writing credentials
            touch /etc/franky.env
            chmod 0600 /etc/franky.env

            # Remove stale agent credentials from previous runs to avoid duplicates
            sed -i '/^FRANKY_AGENT_ID=/d; /^FRANKY_AGENT_SECRET=/d; /^FRANKY_TEAM_ID=/d; /^FRANKY_INBOX_SERVER=/d' /etc/franky.env

            # Store agent credentials using printf to avoid shell expansion
            printf 'FRANKY_AGENT_ID=%s\n' "$FRANKY_AGENT_ID" >> /etc/franky.env
            printf 'FRANKY_AGENT_SECRET=%s\n' "$FRANKY_AGENT_SECRET" >> /etc/franky.env
            printf 'FRANKY_TEAM_ID=%s\n' "$FRANKY_TEAM_ID" >> /etc/franky.env
            printf 'FRANKY_INBOX_SERVER=%s\n' "http://127.0.0.1:8080" >> /etc/franky.env

            echo "Agent credentials written to /etc/franky.env"
            REG_OK=true
        else
            echo "WARNING: failed to parse agent credentials from response: $REG_RESPONSE"
        fi
    else
        echo "WARNING: failed to register agent with franky-box"
    fi
else
    echo "WARNING: franky-box not ready, skipping agent registration"
fi

# Start franky only if registration succeeded (needs agent credentials)
if [ "$REG_OK" = "true" ]; then
    sctl start franky
    sctl enable franky
else
    echo "WARNING: skipping franky start — agent registration failed, no credentials"
fi

# ============================================
# Cleanup
# ============================================
echo "=== Cleaning up ==="
rm -f /tmp/go*.tar.gz /tmp/zig*.tar.xz
apt-get autoremove -y
apt-get clean

echo "=== mini PC setup complete ==="
